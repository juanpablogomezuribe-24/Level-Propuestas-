        function calculateProjections() {

            // Obtener inputs

            const db = parseInt(document.getElementById('input-db').value);

            const intRate = parseFloat(document.getElementById('input-int').value) / 100;

            const convRate = parseFloat(document.getElementById('input-conv').value) / 100;

            const cltv = parseInt(document.getElementById('input-cltv').value);


            // Actualizar etiquetas

            document.getElementById('val-db').innerText = db.toLocaleString('es-CO');

            document.getElementById('val-int').innerText = (intRate * 100).toFixed(0) + '%';

            document.getElementById('val-conv').innerText = (convRate * 100).toFixed(1) + '%';

            document.getElementById('val-cltv').innerText = '$' + cltv.toLocaleString('es-CO');


            // Calcular resultados

            const totalInteractions = Math.round(db * intRate);

            const totalRegisters = Math.round(totalInteractions * convRate);

            const totalROI = totalRegisters * cltv;


            // Actualizar salidas en UI

            document.getElementById('res-contacts').innerText = totalInteractions.toLocaleString('es-CO');

            document.getElementById('res-registers').innerText = totalRegisters.toLocaleString('es-CO');

            document.getElementById('res-roi').innerText = '$' + totalROI.toLocaleString('es-CO');

        }


        // Ejecutar primer cálculo al cargar la página

        window.onload = function() {

            calculateProjections();
